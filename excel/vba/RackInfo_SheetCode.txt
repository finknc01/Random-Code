' Replace only the Worksheet_Change code in Sheet (Rack Info).
Option Explicit

Private Sub Worksheet_Change(ByVal Target As Range)
    Dim table As ListObject
    Dim rackIds As Range, heights As Range

    On Error GoTo HandleError
    If Not Application.EnableEvents Then Exit Sub

    Set table = Me.ListObjects("tblRacks")
    If table.DataBodyRange Is Nothing Then Exit Sub

    Set rackIds = table.ListColumns("Rack ID").DataBodyRange
    Set heights = table.ListColumns("Rack Height").DataBodyRange
    If Application.Intersect(Target, Union(rackIds, heights)) Is Nothing Then Exit Sub

    ' Flag duplicates and heights outside 1..42 without rewriting user data.
    modRackRegistry.ValidateRackRegistry
    modRackElevation.RenderRackElevation
    Exit Sub

HandleError:
    MsgBox "Rack Info event error (" & Err.Number & "): " & _
           Err.Description, vbExclamation
End Sub
