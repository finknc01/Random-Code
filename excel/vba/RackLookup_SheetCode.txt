' Rack Lookup sheet object: retain Worksheet_Change and handle ActiveX buttons.
' Control names on the worksheet: cmdClear, cmdRefresh, cmdAddServer.
' Paste into Microsoft Excel Objects > Sheet (Rack Lookup), NOT a standard module.
Option Explicit

Private Sub Worksheet_Change(ByVal Target As Range)
    On Error GoTo HandleError
    If Not Application.EnableEvents Then Exit Sub
    If Application.Intersect(Target, Me.Range("C3,C4,C6,C8,C10")) Is Nothing Then Exit Sub

    modRackSearch.HandleSearchChange Me, Target
    modRackElevation.RenderRackElevation
    Exit Sub

HandleError:
    MsgBox "Rack Lookup event error (" & Err.Number & "): " & _
           Err.Description, vbExclamation
End Sub

' These Click events run from the ActiveX CommandButtons on Rack Lookup.
' Existing button modules remain the reusable entry points.

Private Sub cmdClear_Click()
    buttonClear.ClearRackLookupButton
End Sub

Private Sub cmdRefresh_Click()
    buttonRefresh.RefreshRackLookupButton
End Sub

Private Sub cmdAddServer_Click()
    buttonAddServer.OpenAddServerForm
End Sub
