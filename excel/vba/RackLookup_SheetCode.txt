' Replace only the Worksheet_Change code in Sheet (Rack Lookup).
Option Explicit

Private Sub Worksheet_Change(ByVal Target As Range)
    On Error GoTo HandleError
    If Not Application.EnableEvents Then Exit Sub
    If Application.Intersect(Target, Me.Range("C3,C4,C6,C8,C10")) Is Nothing Then Exit Sub

    modRackSearch.HandleSearchChange Me, Target
    modRackElevation.RenderRackElevation
    Exit Sub

HandleError:
    MsgBox "Rack Lookup event error (" & Err.Number & "): " & _
           Err.Description, vbExclamation
End Sub
