' Paste this into the code window for the Rack Lookup worksheet.
' In the VBA editor (Alt+F11), double-click SheetN (Rack Lookup).
' Replace the earlier Worksheet_Change event; do not keep two copies.
Option Explicit

Private Sub Worksheet_Change(ByVal Target As Range)
    If Target.CountLarge <> 1 Then Exit Sub
    If Intersect(Target, Me.Range("C3,C4,C6,C8,C10")) Is Nothing Then Exit Sub

    ' First clear conflicting search inputs, preserving rack + U.
    modRackSearch.HandleSearchChange Me, Target

    ' Then draw the full rack and highlight any matched server(s).
    modRackElevation.RenderRackElevation
End Sub
