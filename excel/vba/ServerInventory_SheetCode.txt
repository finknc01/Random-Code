' Put this event in the Server Inventory worksheet code module.
' This is not a standard module and must not replace the Rack Lookup event.
Option Explicit

Private Sub Worksheet_Change(ByVal Target As Range)
    Dim t As ListObject
    Dim rackIdCells As Range

    On Error GoTo HandleError
    Set t = Me.ListObjects("tblInventory")
    If t.DataBodyRange Is Nothing Then Exit Sub
    If Intersect(Target, t.DataBodyRange) Is Nothing Then Exit Sub

    ' Only rescan the rack registry when a Rack ID has changed.
    Set rackIdCells = t.ListColumns("Rack ID").DataBodyRange
    If Not Intersect(Target, rackIdCells) Is Nothing Then
        modRackRegistry.SyncRackRegistry
    End If

    ' Other data edits (hostname, U, serial, equipment height, etc.)
    ' also update the currently displayed rack.
    modRackElevation.RenderRackElevation
    Exit Sub

HandleError:
    MsgBox "Inventory refresh error: " & Err.Description, vbExclamation
End Sub
