' Replace only the Worksheet_Change code in Sheet (Server Inventory).
Option Explicit

Private Sub Worksheet_Change(ByVal Target As Range)
    Dim table As ListObject, rackIds As Range

    On Error GoTo HandleError
    If Not Application.EnableEvents Then Exit Sub

    Set table = Me.ListObjects("tblInventory")
    If Application.Intersect(Target, table.Range) Is Nothing Then Exit Sub

    ' Handles edits and pasted ranges, including newly appended table rows.
    If Not table.DataBodyRange Is Nothing Then
        Set rackIds = table.ListColumns("Rack ID").DataBodyRange
        If Not Application.Intersect(Target, rackIds) Is Nothing Then
            modRackRegistry.SyncRackRegistry
        End If
    End If

    ' Also redraw after edits to U, hostname, serial, asset, or equipment height.
    modRackElevation.RenderRackElevation
    Exit Sub

HandleError:
    MsgBox "Inventory event error (" & Err.Number & "): " & _
           Err.Description, vbExclamation
End Sub
